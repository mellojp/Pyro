from google.colab import drive
import zipfile
import os
import shutil
import glob

# 1. Montar o Drive
drive.mount('/content/drive')

# 2. Caminhos
pasta_drive = '/content/drive/MyDrive/resultados_pesquisa_fogo'
zip_origem = os.path.join(pasta_drive, 'resultados_treino_v4.zip')
zip_dataset = os.path.join(pasta_drive, 'dataset_yolo_processado.zip')
pasta_trabalho = '/content/yolov5'

os.makedirs(pasta_trabalho, exist_ok=True)
os.chdir(pasta_trabalho)

# 3. Extrair os resultados do treino v4 para dentro do yolov5
print("A extrair os ficheiros de treino v4...")
with zipfile.ZipFile(zip_origem, 'r') as zf:
    zf.extractall('.')

# 4. Extrair o dataset DENTRO da pasta yolov5 (para o export.py o encontrar corretamente)
if not os.path.exists('dataset_yolo_processado'):
    print("A extrair dataset para o diretório correto de calibração INT8...")
    with zipfile.ZipFile(zip_dataset, 'r') as zf:
        zf.extractall('dataset_yolo_processado')

# 5. Localizar dinamicamente o best.pt que acabou de ser extraído
pesos = glob.glob('**/weights/best.pt', recursive=True)

if not pesos:
    raise FileNotFoundError("Não foi possível encontrar o ficheiro best.pt dentro do ZIP extraído!")

best_pt = pesos[0]
print(f"Modelo encontrado com sucesso em: {best_pt}")

# 6. Exportar para TFLite INT8 com calibração bem-sucedida
!python export.py \
    --weights {best_pt} \
    --include tflite \
    --imgsz 320 \
    --int8 \
    --data dataset_yolo_processado/data.yaml

# 7. Salvar o resultado INT8 real no Drive
tflite_files = glob.glob('**/weights/*.tflite', recursive=True)

if tflite_files:
    pasta_origem = os.path.dirname(tflite_files[0])
    saida_zip = os.path.join(pasta_drive, 'resultados_treino_v4_int8')
    shutil.make_archive(saida_zip, 'zip', pasta_origem)
    print(f"Sucesso! Modelo INT8 gerado e salvo em: {saida_zip}.zip")
else:
    print("Erro: O ficheiro .tflite quantizado ainda não foi gerado. Verifique os logs acima.")
